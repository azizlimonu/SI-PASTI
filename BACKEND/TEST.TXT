TEST PIHAK
===========================================================================================>
# Get semua pihak
GET http://localhost:3000/api/pihak
Authorization: Bearer [token]

# Search pihak
GET http://localhost:3000/api/pihak?search=Abdul&jenis_pihak=ASN
Authorization: Bearer [token]

# Get detail pihak + riwayat temuan
GET http://localhost:3000/api/pihak/1
Authorization: Bearer [token]

# Tambah pihak baru
POST http://localhost:3000/api/pihak
Authorization: Bearer [token]
Content-Type: application/json
{
  "nama": "Abdul Aziz",
  "nip": "19981031202504001",
  "jabatan": "Auditor Ahli Pertama",
  "instansi_perusahaan": "Inspektorat Kabupaten Gorontalo",
  "jenis_pihak": "ASN"
}

# Cek SKTJM by NIP
GET http://localhost:3000/api/pihak/sktjm?nip=19981031202504001
Authorization: Bearer [token]

# Cek SKTJM by nama
GET http://localhost:3000/api/pihak/sktjm?nama=Abdul
Authorization: Bearer [token]

# Cek SKTJM by pihak_id
GET http://localhost:3000/api/pihak/sktjm?pihak_id=1
Authorization: Bearer [token]
